Rock, paper, scissors, three ways

A program that plays rock paper scissors has to work out who won each round. Today you write that function three ways: with if lines, with a dictionary, and by counting steps around a circle. You play five rounds against the computer, and you check that all three versions give the same answer for every pairing.

What the caller can do. A returned value goes to the line that called the function. That line can print it, store it in a variable, or compare it with something else. The rules: rock beats scissors, scissors beats paper, and paper beats rock. If both players pick the same move, it is a tie.

rps_game.py

"""Session 6: five rounds of rock paper scissors against the computer."""

import random

MOVES = ["rock", "paper", "scissors"]

# each key beats its value: rock beats scissors
BEATS = {"rock": "scissors", "paper": "rock", "scissors": "paper"}

# what to tell you, for each answer winner can return
SAY = {"tie": "it is a tie", "player 1": "you win", "player 2": "the computer wins"}


def winner(move1, move2):
    """Return who wins when player 1 plays move1 and player 2 plays move2."""
    if move1 == move2:
        return "tie"
    elif BEATS[move1] == move2:
        return "player 1"
    else:
        return "player 2"


wins = 0
losses = 0
ties = 0

for turn in range(5):
    my_move = input("rock, paper, or scissors? ")

    while my_move not in MOVES:
        my_move = input("type rock, paper, or scissors: ")

    computer_move = random.choice(MOVES)
    result = winner(my_move, computer_move)
    print(f"the computer played {computer_move}, so {SAY[result]}")

    if result == "player 1":
        wins = wins + 1
    elif result == "player 2":
        losses = losses + 1
    else:
        ties = ties + 1

print(f"you won {wins}, lost {losses}, and tied {ties}")

rps_three_ways.py

"""Session 6: three versions of winner, checked against each other."""

MOVES = ["rock", "paper", "scissors"]

# each key beats its value: rock beats scissors
BEATS = {"rock": "scissors", "paper": "rock", "scissors": "paper"}

# what each number of steps means
RESULTS = ["tie", "player 1", "player 2"]


def winner_if(move1, move2):
    """Return who wins, decided with if lines."""
    if move1 == move2:
        return "tie"
    elif move1 == "rock" and move2 == "scissors":
        return "player 1"
    elif move1 == "scissors" and move2 == "paper":
        return "player 1"
    elif move1 == "paper" and move2 == "rock":
        return "player 1"
    else:
        return "player 2"


def winner_dict(move1, move2):
    """Return who wins, decided with the BEATS dictionary."""
    if move1 == move2:
        return "tie"
    elif BEATS[move1] == move2:
        return "player 1"
    else:
        return "player 2"


def winner_clock(move1, move2):
    """Return who wins, decided by counting steps around the circle of moves."""
    spot1 = MOVES.index(move1)
    spot2 = MOVES.index(move2)

    steps = (spot1 - spot2) % 3
    return RESULTS[steps]


agree = 0

for move1 in MOVES:
    for move2 in MOVES:
        first = winner_if(move1, move2)
        second = winner_dict(move1, move2)
        third = winner_clock(move1, move2)

        if first == second and second == third:
            agree = agree + 1

print(f"the three versions agree on {agree} of the 9 pairings")

rps_spock.py, the stretch

"""Session 6 stretch: rock paper scissors lizard Spock, decided two ways."""

MOVES = ["rock", "paper", "scissors", "spock", "lizard"]

# each move beats the two moves in its list
BEATS = {
    "rock": ["scissors", "lizard"],
    "paper": ["rock", "spock"],
    "scissors": ["paper", "lizard"],
    "spock": ["scissors", "rock"],
    "lizard": ["spock", "paper"],
}


def winner_dict(move1, move2):
    """Return who wins, decided with the BEATS dictionary."""
    if move1 == move2:
        return "tie"
    elif move2 in BEATS[move1]:
        return "player 1"
    else:
        return "player 2"


def winner_clock(move1, move2):
    """Return who wins, decided by counting steps around the circle."""
    spot1 = MOVES.index(move1)
    spot2 = MOVES.index(move2)

    steps = (spot1 - spot2) % 5

    if steps == 0:
        return "tie"
    elif steps % 2 == 1:
        return "player 1"
    else:
        return "player 2"


agree = 0

for move1 in MOVES:
    for move2 in MOVES:
        if winner_dict(move1, move2) == winner_clock(move1, move2):
            agree = agree + 1

print(f"the two versions agree on {agree} of the 25 pairings")
